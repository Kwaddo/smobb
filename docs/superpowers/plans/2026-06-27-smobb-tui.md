```rust
use std::collections::VecDeque;
use std::time::Instant;
use smobb_core::engine::{Engine, Output};
use smobb_core::speaker::Color;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Tab { Options, Journal }

#[derive(Debug, Clone)]
pub struct Span { pub text: String, pub color: Color, pub dim: bool }

pub struct App {
    engine: Engine,
    transcript: Vec<Span>,
    queue: VecDeque<Output>,
    reveal_at: Option<Instant>,
    tab: Tab,
    sidebar_visible: bool,
    scroll: u16,
    input: String,
    awaiting: bool,
    quit: bool,
}

impl App {
    pub fn new(mut engine: Engine) -> App {
        let outputs = engine.start();
        let mut app = App {
            engine,
            transcript: Vec::new(),
            queue: VecDeque::new(),
            reveal_at: None,
            tab: Tab::Options,
            sidebar_visible: true,
            scroll: 0,
            input: String::new(),
            awaiting: false,
            quit: false,
        };
        app.enqueue(outputs);
        app
    }

    fn enqueue(&mut self, outputs: Vec<Output>) {
        self.queue.extend(outputs);
        self.awaiting = false;
        self.reveal_at = None;
    }

    pub fn quit(&self) -> bool { self.quit }
    pub fn transcript(&self) -> &[Span] { &self.transcript }
    pub fn awaiting(&self) -> bool { self.awaiting }
}
```

- Produces: `pub fn tick(&mut self, now: Instant)` — when revealing and `reveal_at` is None or `now >= reveal_at`, pop+apply ONE queued `Output`: `Line`→push opaque `Span`; `Translation`→push dim `Span` `  ⤷ "{t}"`; `Sleep(ms)`→`reveal_at = Some(now + ms)` and return (wait); `Recap(_)`→push dim `Span` `  ✎ noted`; `AwaitInput`→`awaiting = true`, stop; `End`→`quit = true`.
```rust
impl App {
    pub fn tick(&mut self, now: Instant) {
        if self.awaiting || self.quit { return; }
        loop {
            if let Some(at) = self.reveal_at {
                if now < at { return; }
            }
            let item = match self.queue.pop_front() {
                Some(i) => i,
                None => { self.reveal_at = None; return; }
            };
            match item {
                Output::Line { text, color } => {
                    self.transcript.push(Span { text, color, dim: false });
                    self.reveal_at = None;
                }
                Output::Translation(t) => {
                    self.transcript.push(Span {
                        text: format!("  \u{21b7} \"{}\"", t),
                        color: Color::White, dim: true,
                    });
                    self.reveal_at = None;
                }
                Output::Sleep(ms) => {
                    self.reveal_at = Some(now + std::time::Duration::from_millis(ms as u64));
                    return;
                }
                Output::Recap(_) => {
                    self.transcript.push(Span {
                        text: "  \u{270e} noted".to_string(),
                        color: Color::White, dim: true,
                    });
                    self.reveal_at = None;
                }
                Output::AwaitInput => { self.awaiting = true; return; }
                Output::End => { self.quit = true; return; }
            }
        }
    }
}
```

```rust
impl App {
    pub fn submit(&mut self) {
        if !self.awaiting { return; }
        let cmd = std::mem::take(&mut self.input);
        let outputs = self.engine.advance(&cmd);
        self.enqueue(outputs);
    }

    pub fn skip_reveal(&mut self) {
        if self.awaiting || self.quit { return; }
        self.reveal_at = None;
        while let Some(item) = self.queue.pop_front() {
            match item {
                Output::Line { text, color } =>
                    self.transcript.push(Span { text, color, dim: false }),
                Output::Translation(t) =>
                    self.transcript.push(Span { text: format!("  \u{21b7} \"{}\"", t), color: Color::White, dim: true }),
                Output::Recap(_) =>
                    self.transcript.push(Span { text: "  \u{270e} noted".to_string(), color: Color::White, dim: true }),
                Output::Sleep(_) => {} // ignore timing
                Output::AwaitInput => { self.awaiting = true; return; }
                Output::End => { self.quit = true; return; }
            }
        }
    }

    fn type_char(&mut self, c: char) { if self.awaiting { self.input.push(c); } }
    fn backspace(&mut self) { if self.awaiting { self.input.pop(); } }

    #[cfg(test)]
    pub fn submit_for_test(&mut self, s: &str) { self.input = s.to_string(); self.submit(); }
}
```

```rust
use ratatui::crossterm::event::{KeyCode, KeyModifiers};

impl App {
    pub fn key(&mut self, code: KeyCode, mods: KeyModifiers) {
        let ctrl = mods.contains(KeyModifiers::CONTROL);
        match code {
            KeyCode::Char('c') if ctrl => self.quit = true,
            KeyCode::Esc => self.quit = true,
            KeyCode::Enter => self.submit(),
            KeyCode::Backspace => self.backspace(),
            KeyCode::Tab => self.tab = match self.tab { Tab::Options => Tab::Journal, Tab::Journal => Tab::Options },
            KeyCode::F(1) => self.sidebar_visible = !self.sidebar_visible,
            KeyCode::F(2) => { let s = self.engine.skip(); self.engine.set_skip(!s); }
            KeyCode::Up => self.scroll = self.scroll.saturating_sub(1),
            KeyCode::Down => self.scroll = self.scroll.saturating_add(1),
            KeyCode::PageUp => self.scroll = self.scroll.saturating_sub(5),
            KeyCode::PageDown => self.scroll = self.scroll.saturating_add(5),
            KeyCode::Char(c) if !ctrl => {
                if self.awaiting { self.type_char(c); }
                else if c == ' ' { self.skip_reveal(); }
            }
            _ => {}
        }
    }

    pub fn tab(&self) -> Tab { self.tab }
    pub fn sidebar_visible(&self) -> bool { self.sidebar_visible }
    pub fn scroll(&self) -> u16 { self.scroll }
    pub fn input(&self) -> &str { &self.input }
    pub fn options_lines(&self) -> Vec<Output> { self.engine.options_view() }
    pub fn journal_lines(&self) -> Vec<Output> { self.engine.journal_view() }
}
```

```rust
use ratatui::{
    Frame,
    layout::{Constraint, Layout, Rect},
    style::{Color as RColor, Modifier, Style},
    text::{Line, Span as RSpan},
    widgets::{Block, Paragraph, Wrap},
};
use smobb_core::speaker::Color;
use smobb_core::engine::Output;
use crate::app::App;

fn to_rgb(c: Color) -> RColor {
    let hex = c.hex(); // "#rrggbb"
    let r = u8::from_str_radix(&hex[1..3], 16).unwrap_or(255);
    let g = u8::from_str_radix(&hex[3..5], 16).unwrap_or(255);
    let b = u8::from_str_radix(&hex[5..7], 16).unwrap_or(255);
    RColor::Rgb(r, g, b)
}

fn output_lines_to_text(outs: &[Output]) -> Vec<Line<'static>> {
    outs.iter().filter_map(|o| match o {
        Output::Line { text, color } =>
            Some(Line::from(RSpan::styled(text.clone(), Style::default().fg(to_rgb(*color))))),
        _ => None,
    }).collect()
}

pub fn render(f: &mut Frame, app: &App) {
    let outer = Block::bordered().title(" S.M.O.B.B. ");
    let area = f.area();
    let inner = outer.inner(area);
    f.render_widget(outer, area);

    // rows: body (fill) + input (1)
    let rows = Layout::vertical([Constraint::Min(1), Constraint::Length(1)]).split(inner);
    let body = rows[0];
    let input_area = rows[1];

    // body columns: story (70%) + sidebar (30%) when visible
    let (story_area, sidebar_area): (Rect, Option<Rect>) = if app.sidebar_visible() {
        let cols = Layout::horizontal([Constraint::Percentage(70), Constraint::Percentage(30)]).split(body);
        (cols[0], Some(cols[1]))
    } else {
        (body, None)
    };

    // story transcript
    let story_lines: Vec<Line> = app.transcript().iter().map(|s| {
        let mut style = Style::default().fg(to_rgb(s.color));
        if s.dim { style = style.add_modifier(Modifier::DIM); }
        Line::from(RSpan::styled(s.text.clone(), style))
    }).collect();
    let story = Paragraph::new(story_lines)
        .wrap(Wrap { trim: false })
        .scroll((app.scroll(), 0))
        .block(Block::bordered());
    f.render_widget(story, story_area);

    // sidebar
    if let Some(sa) = sidebar_area {
        let (title, outs) = match app.tab() {
            crate::app::Tab::Options => (" OPTIONS ", app.options_lines()),
            crate::app::Tab::Journal => (" JOURNAL ", app.journal_lines()),
        };
        let side = Paragraph::new(output_lines_to_text(&outs))
            .wrap(Wrap { trim: false })
            .block(Block::bordered().title(title));
        f.render_widget(side, sa);
    }

    // input line
```

```rust
use std::time::{Duration, Instant};
use ratatui::crossterm::event::{self, Event, KeyEventKind};

fn main() -> std::io::Result<()> {
    let mut terminal = ratatui::init(); // raw mode + alt screen + panic hook
    let mut app = App::new(Engine::new(content::load()));

fn run(terminal: &mut ratatui::DefaultTerminal, app: &mut App) -> std::io::Result<()> {
    let tick = Duration::from_millis(30);
    while !app.quit() {
        terminal.draw(|f| ui::render(f, app))?;
        if event::poll(tick)? {
            if let Event::Key(k) = event::read()? {
                if k.kind == KeyEventKind::Press {
                    app.key(k.code, k.modifiers);
                }
            }
        }
        app.tick(Instant::now());
    }
    Ok(())
}
```
